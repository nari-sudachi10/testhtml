センサーから値を取得する

AHT20+BMP280/BME2803 in 1というセンサーを使って部屋の気温と湿度と気圧を取得します

1.IC2デバイスの確認

まずセンサーが認識できるか確認するためにIC2スキャナを作成します


#include 

#define SDA_PIN 4
#define SCL_PIN 5

void setup() {
  Serial.begin(115200);
  delay(1000);

  Wire.begin(SDA_PIN, SCL_PIN);

  Serial.println("I2C Scanner");
}

void loop() {
  Serial.println("Scanning...");

  int count = 0;

  for (byte address = 1; address < 127; address++) {
    Wire.beginTransmission(address);

    byte error = Wire.endTransmission();

    if (error == 0) {
      Serial.print("I2C device found at 0x");

      if (address < 16) {
        Serial.print("0");
      }

      Serial.println(address, HEX);
      count++;
    }
  }

  if (count == 0) {
    Serial.println("No I2C devices found.");
  }

  Serial.println("-----");

  delay(3000);
}

プログラムを保存したあと、メニューバーにある書き込みボタンを押してください。 その後、ツール→シリアルモニターを開いてください。通信速度を115200にすると結果が出てくるはずです 


	Scanning...
	I2C device found at 0x38
	I2C device found at 0x77

こんな感じに表示されていたらOK！
＊もしうまく表示されない場合ESP32にあるEN/RSTボタンを押してみてください
