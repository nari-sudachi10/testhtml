# Gitの状態を確認 
$ git status
# ファイルをステージング 
$ git add ファイル名
$ git add .
# コミット 
$ git commit -m "コミットメッセージ"
# GitHubへプッシュ 
$ git push
# GitHubから変更を取得 
$ git pull
# ブランチ一覧 
$ git branch
# ブランチ作成 
$ git branch ブランチ名
# ブランチ切り替え 
$ git switch ブランチ名
# ブランチ作成 + 切り替え 
$ git switch -c ブランチ名
# リモートリポジトリを確認 
$ git remote -v
# リポジトリをクローン 
$ git clone URL
# コミット履歴を確認 
$ git log
# 直前のコミットを確認しやすく表示 
$ git log --oneline
# GitHubの変更を反映するとき 
$ git pull
